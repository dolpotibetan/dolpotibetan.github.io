༄༅། །ཡོ་ཧ་ནན་​གྱིས་​བྲིས་པའི་ཡི་གེ་​གསུམ་པ​། །
ལེའུ་དང་པོ།
འཚམས་འདྲི།
1 ཆོས་ཚོགས་ཀྱི་རྒན་མི་ངས་ཡི་གེ་འདི་ནི་བདེན་པའི་བརྒྱུད་ནས་ངས་བརྩེ་བ་བགྱིད་ས་ག་ཡུ་ཁོའི་ཆེད་དུ་འབྲི་འོད།
2 ངའི་བརྩེ་བའི་རོགས་པོ་ཁྱོད་ཀྱི་སེམས་ལེ་བཞི་བདེ་འོད་དི་ཅོག་འདྲ། ཁྱོད་ཀྱི་ལུས་ཁམས་ཟང་བདེ་མོ་འོད་དི་དིང་། ཡང་ཁྱོད་ཀྱིས་ལས་དོན་ཡེ་རེ་འགྲུབ་འོད་དི་རེ་བ་བགྱིད་འོད།
3 ཡང་དེ་སྔོན་ཆོས་གྲོགས་སྤུན་ཟླ་ཁ་ཤས་འདི་རུ་སླེབས་ཏེ། ཁྱོད་རང་བདེན་པའི་ལམ་ལེ་འགྲོ་འོད་དི་མ་ཟད། བདེན་པ་དིང་ཅོག་འདྲ་སྤྱོད་ལམ་དགའ་མོ་བགྱིད་འོད་དི་ཁྱོད་ཀྱི་སྐོར་ལེ་ལབ་ཐོས་ཏེ་ང་དགའ་རི་ཆེན་པོ་འཇུ་གུ་བྱུང་།
4 ངའི་དད་སེམས་གཅིག་པའི་གཞོན་ནུ་ཁྱེད་ཀྱ་བདེན་པའི་ལམ་ལེ་མུ་མཐུད་ནས་འགྲོ་འོད་དི་གནས་ཚུལ་ཐོས་ཕེན། དེ་ལས་ལྷག་གི་དགའ་སྤྲོ་ཆེ་བ་ཞིག་ང་ལེ་མེད།
དཀོན་མཆོག་གི་ཞབས་ཕྱི་ལེ་རོགས་ལམ་བགྱིད་དགོས་པའི་སྐོར།
5 ངའི་བརྩེ་བའི་རོགས་པོ། སྤུན་ཟླ་དེ་ཀྱ་ནི། ཁྱོད་ཀྱིས་ངོ་མི་ཤེས་འོད་ནའང་། དད་པའི་ངང་ནས་རོགས་རམ་བྱས་པི་དེ་ནི། ངོ་ཐོག་སེམས་བཟང་པོ་ཞིག་ཡིན་དིག
6 སྤུན་ཟླ་དེ་ཀྱ་ཀྱིས་ཟང་ཆོས་ཚོགས་ཀྱི་མདུན་དུ། ཁྱོད་ཀྱིས་བྱས་པའི་བྱམས་སེམས་དེ་ལེ་དཔང་པོ་བྱས་པི་དིག སྤུན་ཟླ་དེ་ཀྱ་ནི་དཀོན་མཆོག་གི་ཞབས་ཕྱི་བགྱིད་ཆོག་འོས་པ་ཞིག་ཡིན་ཕེན། དཀོན་མཆོག་གི་ཐུགས་དགྱེས་རྒྱུ་དོན་ལེ། ཁོང་ཀྱ་འགྲིམས་འགྲུལ་བགྱིད་རྒྱུ་དོན་ལེ་རོགས་རམ་བྱས་ནི་དགའ་མོ་འོད་དིག
7 ཅི་ལེ་ཟེར་ནི། ཁོང་ཀྱ་ནི་དད་པ་མི་བགྱིད་མཁན་ཀྱ་ཀྱི་ལག་ནས་རོགས་སྐྱོར་ཅི་གང་མ་བླངས་ཏེ། སྐྱབས་མགོན་ཡེ་ཤུའི་དོན་ལེ་ཕྱག་ལས་བགྱིད་རྒྱུ་ཆེད་དུ་ཕྱི་ལེ་བུད་དི་དིག
8 འོ་དེ་ལེ་བརྟེན་ཏེ། འུ་ཀྱ་བདེན་པའི་དོན་ལེ་ལས་རོགས་བགྱིད་རྒྱུ་ཆེད་དུ། འུ་ཀྱ་ཀྱིས་ཟང་མི་དེ་ཅོག་འདྲ་ཀྱ་ལེ་རྒྱབ་སྐྱོར་བགྱིད་དགོས་འོད་དིག
9 སྔོན་མ་ངས་ཆོས་ཚོགས་ལེ་ཡི་གེ་ཞིག་བྲིས་པི་ཡིན། ཡིན་ནའང་། ཆོས་ཚོགས་ཀྱི་ནང་ལེ་རང་ཆེ་འདོད་བགྱིད་མཁན་དིའོ་ཏྲེ་ཕེ་ཁོས། ངའི་དབང་ཆ་དེ་ལེ་ཁས་མ་བླངས་པི་དིག
10 འོ་དེ་ལེ་བརྟེན་ཏེ། ང་རང་འོང་སྟེ་ཁོས་ཅི་བྱས་པའི་སྐོར་ལེ་མངོན་སུམ་དུ་ཐུག་སྟེ་སྐད་ཆ་ཤོད་ཀྱི་ཡིན། ཁོས་ཉེད་ཀྱ་ལེ་དམའ་འབེབས་གཏོང་གྱོ་འོད་དི་མ་ཟད། ད་རུང་འདོད་པ་མ་ཚིམས་ཏེ། ཁོས་ཆོས་གྲོགས་སྤུན་ཟླ་ཀྱ་ལེ་ཕེབས་བསུ་མི་ཞུ་འོད་དིག ཡང་ཕེབས་བསུ་ཞུ་འདོད་འོད་ཡོད་དེ་ཀྱ་ལེ་ཟང་བཀའ་འགོག་བགྱིད་འོད་དི་མ་ཟད། ཕེབས་བསུ་ཞུ་འདོད་བགྱིད་མཁན་ཀྱ་ཟང་ཆོས་ཚོགས་ནས་ཕྱི་བུད་གཏོང་གྱོ་འོད་དིག
11 ངའི་བརྩེ་བའི་རོགས་པོ་ཁྱོད་ཀྱིས་མིག་དཔེ་བཟང་པོ་ལེ་མ་གཏོགས་ཞིག་ལེ། མིག་དཔེ་ངན་པ་ལེ་མ་ལྟས་ཤིག ལས་བཟང་པོ་བགྱིད་མཁན་དེ་དཀོན་མཆོག་ནས་ཡིན་དིག ཡང་ལས་ངན་པ་བགྱིད་མཁན་དེས་དཀོན་མཆོག་ངོ་མི་ཤེས་འོད་དིག
12 ཡང་དེ་མེ་ཏྲི་ཡུ་ཁོའི་སྐོར་ལེ་མི་ཡེ་རེས་བསྟོད་ར་བཏང་པི་མ་ཟད། ཉེད་ཀྱ་ཀྱིས་ཟང་ཁོ་ལེ་བསྟོད་ར་གཏོང་གྱོ་འོད། ཁོས་རང་གི་སྤྱོད་ལམ་བཟང་པོ་བྱས་པི་དེས་བདེན་པའི་དཔང་པོ་བགྱིད་འོད་དིག ཡང་ཉེད་ཀྱ་ཀྱིས་ཟང་ཁོ་ལེ་བསྟོད་ར་གཏོང་གྱོ་འོད། ཉེད་ཀྱ་ཀྱིས་བཏང་བའི་བསྟོད་ར་དེ་ནི། བདེན་པའི་དཔང་པོ་ཡིན་དི་ཁྱོད་ཀྱིས་ཧ་གོ་འོད་དིག
མཇུག་མཐའ་མའི་འཚམས་འདྲི།
13 ཁྱོད་ཀྱི་ཆེད་དུ་འབྲི་རྒྱུ་མང་པོ་འོད། ཡིན་ནའང་། སྣག་ཚ་དིང་སྙུག་གུའི་བརྒྱུད་ནས་པ་རྐྱང་འབྲི་འདོད་མེད།
14 ང་རང་གང་མྱུར་མྱུར་ཁྱོད་དིང་གདོང་ངོ་མངོན་སུམ་ཐུག་སྟེ་སྐད་ཆ་ཤོད་རྒྱུ་རེ་བ་བགྱིད་འོད།
15 ད་ཁྱོད་ལེ་དཀོན་མཆོག་གི་ཞི་བདེ་འཐོབ་པར་ཤོག མཛའ་རོགས་ཀྱ་ཀྱིས་ཁྱོད་ལེ་འཚམས་འདྲི་བཀྲ་ཤིས་བདེ་ལེགས་ཞུ་འདུག ཡང་ཁྱོད་དིང་སྐྱེན་ལེ་འོད་ཡོད་རོགས་པོ་ཀྱ་ལེ་ངའི་ཚབ་བྱས་ཏེ། ཁོང་ཀྱ་ཀྱི་མིང་རེ་རེ་ནས་འབོད་དེ་འཚམས་འདྲི་བཀྲ་ཤིས་བདེ་ལེགས་ཞུ་རོགས། ། ། །
