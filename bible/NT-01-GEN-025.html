1 དེ་ནས་ཡང་ཨབ་ར་ཧམ་གྱིས་ཀེ་ཏི་རཱ་ཟེར་མཁན་བུད་མེད་དེ་རང་གི་མནའ་མ་ལེ་བླངས་ཕེན། 2 མོ་ལེ་བི་ས་ཛིམ་རཱན། ཡོག་ཤན། མེ་དན། མིད་ཡཱན། ཡེས་བག་དིང་། ཤུ་ཨག་ཟེར་མཁན་དྲུག་པོ་དེ་ཀྱ་སྐྱེས་པི་དིག 3 ཡོག་ཤན་ནི། བི་ས་ཤེ་བཱ་དིང་དེ་དཱན་ཟེར་མཁན་ཁོ་གཉིས་ཨ་ཀྱི་ཡིན་དིག བི་ས་དེ་དཱན་གྱི་རྒྱུད་པ་ནས་ཨ་ཤུར་ཟེར་རི་མི་རིགས་དིང་། ལཱ་ཐུ་ཤིམ་ཟེར་རི་མི་རིགས། ལཱ་ཨུམ་ཟེར་རི་མི་རིགས་དེ་ཀྱ་བྱུང་པི་དིག 4 ཡང་མིད་ཡཱན་ནི། ཨེ་ཕཱ་དིང་། ཨེ་ཕེར། ཧ་ནོག ཨ་བི་དཱ། ཨེལ་དཱ་ཨ་ཟེར་མཁན་བི་ས་ལྔ་པོ་དེ་ཀྱ་ཀྱི་ཨ་ཀྱི་ཡིན་དིག བི་ས་དེ་ཀྱ་ནི་ཀེ་ཏི་རཱ་ཟེར་མཁན་གྱི་རིགས་རྒྱུད་ཡིན་དིག 5 ཨབ་ར་ཧམ་གྱིས་རང་གི་རྒྱུ་ནོར་ཡེ་རེ་བི་ས་ཨི་སཱག་ལེ་བྱིན་དི་མ་ཟད། 6 རང་གི་ཆུང་མ་གཞན་མ་ཀྱ་ཀྱི་བི་ས་ལེ་ཟང་རྒྱུ་ནོར་བྱིན་དི་དིག དེ་ནས་ཨབ་ར་ཧམ་མ་ཤི་བའི་སྔོན་ལེ། བི་ས་གཞན་མ་དེ་ཀྱ་ནི་བི་ས་ཨི་སཱག་དིང་ཁ་བྲལ་ཏེ་ཤར་ཕྱོགས་གི་ཡུལ་ལེ་འགྲོ་རུ་བཅུག་པི་དིག

27 དེ་ནས་བི་ས་གཉིས་པོ་དེ་ཆེན་པོ་འཚར་ཏེ། ཨེ་སཱབ་ཁོ་ནི། རི་དྭགས་རྒྱག་མཁས་པོ་འོད་ཡོད་མི་ཞིག་བྱུང་སྟེ། ཁོ་ནི་ནམ་རྒྱུན་ཅོག་འདྲ། རི་དྭགས་རྒྱག་ཏུ་འགྲོ་འོད་དིག ཡ་ཀོབ་ཁོ་ནི། གཤིས་ཀ་ཁུ་སིམ་པོ་འོད་ཡོད་ཞིག་ཡིན་དི་མ་ཟད། ནམ་རྒྱུན་དུ་གུར་གྱི་ནང་ལེ་སྡོད་གྱོ་འོད་དིག 28 ཡང་ཨི་སཱག་གིས་ཟང་བི་ས་ཨེ་སཱབ་ལེ་གཅེས་པོ་བགྱིད་འོད་དིག ཅི་ལེ་ཟེར་ནི། ཁོ་ནི་བི་ས་ཨེ་སཱབ་ཀྱིས་རི་དཱགས་བསད་དེ་ཁྱེར་འོང་བའི་ཤ་ལེ་འཐད་པོ་བགྱིད་འོད་དིག ཡིན་ནའང་། རིབ་ཀཱ་མོས་བི་ས་ཆུང་ང་ཡ་ཀོབ་ལེ་གཅེས་པོ་བགྱིད་འོད་དིག 29 དེ་ཡང་ཉིན་མ་གཅིག་ཡ་ཀོབ་ཀྱིས་སྲན་མ་དམར་པོའི་ཐུག་པ་བསྐོལ་གྱི་བསྡད་འོད་དིག དེ་ནས་ཨེ་སཱབ་རི་དྭགས་བརྒྱབ་སྟེ་ལོག་འོང་རྒྱ། ཁོ་གྲོད་པ་འཇུ་གུ་ལྟོགས་འོད་ཕེན། 30 ཁོས་ཡ་ཀོབ་ལེ། ང་གྲོད་པ་འཇུ་གུ་ལྟོགས་ཏེ་ཤི་རྒྱུ་གྲགས། ཁྱོད་ཀྱིས་བསྐོལ་ལི་སྲན་མ་དམར་པོའི་ཐུག་པ་དེ་ནས་ང་ལེ་ཟང་སྦྱིན་དང་ཟིན་སྨྲས་པི་དིག (འོ་དེ་ལེ་བརྟེན་ཏེ། ཁོའི་མིང་ལེ་ཨེ་དོམ་ཟིན་འོད་དིག ཨེ་དོམ་ཟེར་རི་དོན་དེ་ནི། དམར་པོ་ཡིན་དིག) 31 དེ་ནས་ཡ་ཀོབ་ཀྱིས་ཨ་ཇོ་ཨེ་སཱབ་ལེ། གལ་སྲིད། ཁྱོད་ཀྱིས་རང་ལེ་འོད་ཡོད་བི་ས་ཆེ་བའི་ཐོབ་ཐང་དེ་ཐུག་པ་འདི་རིང་བརྗེ་ཚོང་བརྒྱབ་ནི། ངས་ཁྱོད་ལེ་ཐུག་པ་འདི་སྟེར་རོ་ཟིན་སྨྲས་ཕེན། 32 ཨེ་སཱབ་ཀྱིས་ད་ལྟ་ང་འཇུ་གུ་ལྟོགས་ཏེ་ཤི་ལྷག་འདུག ང་ལེ་འོད་ཡོད་བི་ས་ཆེ་བའི་ཐོབ་ཐང་དེ་ཕན་ཐོག་ཅི་འོད་དི་གོ་ཟིན་སྨྲས་པི་དིག 33 ཡང་ཡ་ཀོབ་ཀྱིས་ཁོ་ལེ། ད་ཁྱེད་རང་ལེ་འོད་ཡོད་བི་ས་ཆེ་བའི་ཐོབ་ཐང་དེ་བརྗེ་ཚོང་རྒྱག་ཐགས་གཅོད་ཡིན་ནི། ཁྱོད་ཀྱིས་ང་ལེ་མནའ་སྐྱོལ་དང་ཟིན་སྨྲས་ཕེན། ཨེ་སཱབ་ཀྱིས་ཡ་ཀོབ་ལེ་མནའ་བསྐྱལ་ཏེ། རང་ལེ་འོད་ཡོད་བི་ས་ཆེ་བའི་ཐོབ་ཐང་དེ་ཡ་ཀོབ་ལེ་བཙོངས་པི་དིག 34 དེ་ནས་ཡ་ཀོབ་ཀྱིས་ཨེ་སཱབ་ལེ། སྒོར་ཕྱེ་དིང་སྲན་མའི་ཐུག་པ་བྱིན་ཕེན། ཨེ་སཱབ་ཀྱིས་བཟའ་འཐུང་བྱས་ཏེ་བུད་དི་དིག འོ་འདི་གི་བྱས་ཏེ། ཨེ་སཱབ་ཀྱིས་རང་ལེ་འོད་ཡོད་བི་ས་ཆེ་བའི་ཐོབ་ཐང་དེ་ལེ་མཐོང་ཆུང་བྱས་པི་དིག །
